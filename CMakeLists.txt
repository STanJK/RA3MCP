cmake_minimum_required(VERSION 3.20)
project(RA3MCP LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
option(RA3_BUILD_BRIDGE "Build Windows x86 ASI plugin" ${WIN32})
add_executable(ra3_core_tests native/tests/core_tests.cpp)
target_include_directories(ra3_core_tests PRIVATE native/include)
add_test(NAME core COMMAND ra3_core_tests)
if(RA3_BUILD_BRIDGE)
  if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "The bridge requires MSVC Win32/x86; configure with -A Win32")
  endif()
  set(MINHOOK_DIR "${CMAKE_SOURCE_DIR}/third_party/minhook" CACHE PATH "Extracted MinHook v1.3.4 source root")
  foreach(required include/MinHook.h src/buffer.c src/hook.c src/trampoline.c src/hde/hde32.c)
    if(NOT EXISTS "${MINHOOK_DIR}/${required}")
      message(FATAL_ERROR "Missing ${MINHOOK_DIR}/${required}. See docs/DOWNLOADS.md")
    endif()
  endforeach()
  add_library(minhook STATIC "${MINHOOK_DIR}/src/buffer.c" "${MINHOOK_DIR}/src/hook.c"
    "${MINHOOK_DIR}/src/trampoline.c" "${MINHOOK_DIR}/src/hde/hde32.c")
  target_include_directories(minhook PUBLIC "${MINHOOK_DIR}/include")
  add_library(RA3InputBridge SHARED native/src/bridge.cpp native/bridge.def)
  target_include_directories(RA3InputBridge PRIVATE native/include)
  target_link_libraries(RA3InputBridge PRIVATE minhook bcrypt winmm)
  target_compile_definitions(RA3InputBridge PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE _WIN32_WINNT=0x0601)
  target_compile_options(RA3InputBridge PRIVATE /W4 /permissive- /EHsc /utf-8)
  set_target_properties(RA3InputBridge PROPERTIES PREFIX "" SUFFIX ".asi")
  foreach(t minhook RA3InputBridge ra3_core_tests)
    set_property(TARGET ${t} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  endforeach()
endif()
